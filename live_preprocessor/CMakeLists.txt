cmake_minimum_required(VERSION 3.16)
project(aletheia_live_preprocessor)

find_package(JPEG REQUIRED)
find_package(Threads REQUIRED)
add_library(aletheia_video_frames STATIC src/video_frames.cpp src/video_pipeline.cpp src/video_io.cpp)
target_compile_features(aletheia_video_frames PUBLIC cxx_std_17)
target_include_directories(aletheia_video_frames PUBLIC src)
target_link_libraries(aletheia_video_frames PUBLIC JPEG::JPEG Threads::Threads)

option(ALETHEIA_NATIVE_TESTS_ONLY "Build pure video tests without ROS or the vehicle SDK" OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(aletheia_video_frames_test test/video_frames_test.cpp)
  target_link_libraries(aletheia_video_frames_test aletheia_video_frames)
  add_test(NAME video_frames COMMAND aletheia_video_frames_test)
  add_executable(aletheia_video_io_test test/video_io_test.cpp)
  target_link_libraries(aletheia_video_io_test aletheia_video_frames)
  add_test(NAME video_io COMMAND aletheia_video_io_test)
endif()
if(ALETHEIA_NATIVE_TESTS_ONLY)
  return()
endif()

find_package(ament_cmake REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)

if(BUILD_TESTING)
  add_executable(aletheia_pose_selection_test test/pose_selection_test.cpp)
  target_compile_features(aletheia_pose_selection_test PRIVATE cxx_std_17)
  target_include_directories(aletheia_pose_selection_test PRIVATE src)
  ament_target_dependencies(aletheia_pose_selection_test geometry_msgs tf2)
  add_test(NAME pose_selection COMMAND aletheia_pose_selection_test)
endif()

add_executable(aletheia_live_cloud src/live_cloud_preprocessor.cpp)
target_compile_features(aletheia_live_cloud PUBLIC cxx_std_17)
ament_target_dependencies(aletheia_live_cloud
  geometry_msgs livox_ros_driver2 nav_msgs rclcpp sensor_msgs tf2 tf2_ros)

# 不链接系统 GStreamer：该 sidecar 仅通过受控私有 runtime 的 gst-launch
# 子进程写入原始帧，因此完整 DEB 不依赖目标车的开发包或媒体运行时。
add_executable(aletheia_video_ingest src/video_ingest.cpp)
target_compile_features(aletheia_video_ingest PUBLIC cxx_std_17)
ament_target_dependencies(aletheia_video_ingest rclcpp sensor_msgs)
target_link_libraries(aletheia_video_ingest aletheia_video_frames ${CMAKE_DL_LIBS})

install(TARGETS aletheia_live_cloud aletheia_video_ingest DESTINATION lib/${PROJECT_NAME})
ament_package()
